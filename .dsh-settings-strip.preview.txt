63	@OptIn(ExperimentalMaterial3Api::class)
64	@Composable
65	fun SettingsScreen(viewModel: EropicViewModel) {
66	    val context = LocalContext.current
67	    val settings by viewModel.settings.collectAsStateWithLifecycle()
68	    val count by viewModel.count.collectAsStateWithLifecycle()
69	    val totalSize by viewModel.totalSize.collectAsStateWithLifecycle()
70	    val indexState by viewModel.indexState.collectAsStateWithLifecycle()
71	    val thumbnailCacheSize by viewModel.thumbnailCacheSize.collectAsStateWithLifecycle()
72	    var showRebuildDialog by remember { mutableStateOf(false) }
73	
74	    // 「所有文件访问」是系统设置页授权的，回到应用时要重新读一次真实状态。
75	    val lifecycleOwner = LocalLifecycleOwner.current
76	    var hasWriteAccess by remember { mutableStateOf(viewModel.hasAllFilesAccess()) }
77	    DisposableEffect(lifecycleOwner) {
78	        val observer = LifecycleEventObserver { _, event ->
79	            if (event == Lifecycle.Event.ON_RESUME) {
80	                hasWriteAccess = viewModel.hasAllFilesAccess()
81	            }
82	        }
83	        lifecycleOwner.lifecycle.addObserver(observer)
84	        onDispose { lifecycleOwner.lifecycle.removeObserver(observer) }
85	    }
86	
87	    // 进设置页时刷新一次缓存占用（目录扫描，动作用户可见时才做）。
88	    LaunchedEffect(Unit) { viewModel.refreshThumbnailCacheSize() }
89	    val versionName = remember {
90	        runCatching {
91	            context.packageManager.getPackageInfo(context.packageName, 0).versionName
92	        }.getOrNull() ?: "0.1.0"
93	    }
94	
95	    Column(modifier = Modifier.fillMaxSize()) {
96	        TopAppBar(title = { Text("设置") })
97	        Column(
98	            modifier = Modifier
99	                .fillMaxSize()
100	                .verticalScroll(rememberScrollState())
101	                .padding(horizontal = 12.dp, vertical = 8.dp),
102	            verticalArrangement = Arrangement.spacedBy(16.dp),
103	        ) {
104	            // v0.3：每个分组前加小标题（titleSmall），卡片间距 16dp，分组之间不再糊成一片。
105	            SettingsGroupTitle("外观")
106	            AppearanceCard(settings = settings, viewModel = viewModel)
107	
108	            SettingsGroupTitle("图库与查看")
109	            GalleryViewCard(settings = settings, viewModel = viewModel)
110	
111	            SettingsGroupTitle("清理")
112	            CleanupSettingsCard(settings = settings, viewModel = viewModel)
113	
114	            SettingsGroupTitle("联网识图")
115	            OnlineImageSearchCard(settings = settings, viewModel = viewModel)
116	
117	            SettingsGroupTitle("扫描范围")
118	            ScanScopeCard(settings = settings, viewModel = viewModel)
119	
120	            SettingsGroupTitle("后台与电量")
121	            BackgroundCard(settings = settings, viewModel = viewModel)
122	
123	            SettingsGroupTitle("存储权限")
124	            StorageAccessCard(
125	                hasWriteAccess = hasWriteAccess,
126	                onOpenPermission = { viewModel.openScreen(AppScreen.Permission) },
127	            )
128	
129	            SettingsGroupTitle("缩略图缓存")
130	            ThumbnailCacheCard(
131	                settings = settings,
132	                cacheSizeBytes = thumbnailCacheSize,
133	                viewModel = viewModel,
134	            )
135	
136	            SettingsGroupTitle("本地索引")
137	            Card(modifier = Modifier.fillMaxWidth()) {
138	                Column(modifier = Modifier.padding(14.dp), verticalArrangement = Arrangement.spacedBy(6.dp)) {
139	                    Text(text = "已索引 " + formatCount(count) + " 张 · " + formatBytes(totalSize))
140	                    Text(
141	                        text = "任务状态：" + indexState.phase.name,
142	                        style = MaterialTheme.typography.bodySmall,
143	                        color = MaterialTheme.colorScheme.onSurfaceVariant,
144	                    )
145	                    if (indexState.isRunning) {
146	                        OutlinedButton(onClick = viewModel::cancelIndex) { Text("取消当前任务") }
147	                    } else {
148	                        Button(onClick = viewModel::startIndex) { Text("增量扫描") }
149	                    }
150	                    OutlinedButton(onClick = { showRebuildDialog = true }) { Text("清空并重建索引") }
151	                }
152	            }
153	
154	            SettingsGroupTitle("更多页面")
155	            MorePagesCard(viewModel = viewModel)
156	
157	            SettingsGroupTitle("关于")
158	            Card(modifier = Modifier.fillMaxWidth()) {
159	                Column(modifier = Modifier.padding(14.dp), verticalArrangement = Arrangement.spacedBy(6.dp)) {
160	                    Text(text = "Eropic " + versionName + " · 本地优先 · 仅「联网搜图」时联网")
161	                    Text(
162	                        text = "开源地址：github.com/33kurumi/Eropic",
163	                        style = MaterialTheme.typography.bodySmall,
164	                        color = MaterialTheme.colorScheme.onSurfaceVariant,
165	                    )
166	                    Text(
167	                        text = "没有删除按钮：所有会改动文件的操作都必须由你确认。",
168	                        style = MaterialTheme.typography.bodySmall,
169	                        color = MaterialTheme.colorScheme.onSurfaceVariant,
170	                    )
171	                    OutlinedButton(onClick = { viewModel.openScreen(AppScreen.About) }) {
172	                        Text("权限、数据与许可详情")
173	                    }
174	                }
175	            }
176	        }
177	    }
178	
179	    if (showRebuildDialog) {
180	        AlertDialog(
181	            onDismissRequest = { showRebuildDialog = false },
182	            title = { Text("清空并重建索引？") },
183	            text = {
184	                Text(
185	                    "会删除本地索引与查重结果并重新扫描。不会改动你手机里的任何照片。",
186	                )
187	            },
188	            confirmButton = {
189	                TextButton(onClick = {
190	                    showRebuildDialog = false
191	                    viewModel.rebuildIndex()
192	                }) { Text("重建") }
193	            },
194	            dismissButton = {
195	                TextButton(onClick = { showRebuildDialog = false }) { Text("取消") }
196	            },
197	        )
198	    }
199	}

214	/**
215	 * 联网识图：SauceNAO 的 API Key。
216	 *
217	 * 只有 SauceNAO 需要 key（免费注册后拿到）；Google Lens / Yandex / ascii2d 不填也能用。
218	 * key 存在本地 DataStore，只在请求 SauceNAO 时带上。
219	 */
220	@Composable
221	private fun OnlineImageSearchCard(settings: AppSettings, viewModel: EropicViewModel) {
222	    val context = LocalContext.current
223	    var showKey by remember { mutableStateOf(false) }
224	    // 草稿随已保存的 key 同步：保存成功后 settings 变化，输入框会换成新值。
225	    var keyDraft by remember(settings.sauceNaoApiKey) { mutableStateOf(settings.sauceNaoApiKey) }
226	
227	    Card(modifier = Modifier.fillMaxWidth()) {
228	        Column(modifier = Modifier.padding(14.dp), verticalArrangement = Arrangement.spacedBy(10.dp)) {
229	            Text(
230	                text = "SauceNAO 的接口需要免费的 API Key（在 saucenao.com 注册后获取）。" +
231	                    "不填也能用 Google Lens / Yandex / ascii2d。",
232	                style = MaterialTheme.typography.bodySmall,
233	                color = MaterialTheme.colorScheme.onSurfaceVariant,
234	            )
235	
236	            OutlinedTextField(
237	                value = keyDraft,
238	                onValueChange = { keyDraft = it },
239	                label = { Text("SauceNAO API Key") },
240	                placeholder = { Text("粘贴 saucenao.com 给你的 key") },
241	                singleLine = true,
242	                visualTransformation = if (showKey) {
243	                    VisualTransformation.None
244	                } else {
245	                    PasswordVisualTransformation()
246	                },
247	                modifier = Modifier.fillMaxWidth(),
248	            )
249	
250	            Row(
251	                modifier = Modifier.fillMaxWidth(),
252	                verticalAlignment = Alignment.CenterVertically,
253	                horizontalArrangement = Arrangement.spacedBy(8.dp),
254	            ) {
255	                Button(
256	                    onClick = { viewModel.setSauceNaoApiKey(keyDraft) },
257	                    enabled = keyDraft.trim() != settings.sauceNaoApiKey,
258	                ) { Text("保存") }
259	                TextButton(onClick = { showKey = !showKey }) {
260	                    Text(if (showKey) "隐藏 Key" else "显示 Key")
261	                }
262	            }
263	
264	            // 文字按钮：直接跳到 SauceNAO 主页去注册/登录拿 key。
265	            TextButton(
266	                onClick = {
267	                    runCatching {
268	                        context.startActivity(
269	                            Intent(Intent.ACTION_VIEW, Uri.parse("https://saucenao.com/")),
270	                        )
271	                    }
272	                },
273	            ) { Text("打开 saucenao.com") }
274	
275	            Text(
276	                text = if (settings.sauceNaoApiKey.isEmpty()) {
277	                    "当前未配置 SauceNAO Key。"
278	                } else {
279	                    "当前已配置（" + settings.sauceNaoApiKey.length + " 个字符）。"
280	                },
281	                style = MaterialTheme.typography.labelSmall,
282	                color = MaterialTheme.colorScheme.onSurfaceVariant,
283	            )
284	            if (keyDraft.trim() != settings.sauceNaoApiKey) {
285	                Text(
286	                    text = "有未保存的修改。",
287	                    style = MaterialTheme.typography.labelSmall,
288	                    color = MaterialTheme.colorScheme.tertiary,
289	                )
290	            }
291	        }
292	    }
293	}

700	/**
701	 * 把 SAF 选中的目录 URI 转成「相对存储根」的路径，例如 primary:Pictures/Camera → Pictures/Camera。
702	 *
703	 * 返回 null 表示选中的不是主存储（外置 SD 卡/云端等）；返回空串表示选中的就是存储根本身。
704	 * 只读一次性的 documentId，不保存 URI，所以不需要 takePersistableUriPermission。
705	 */
706	private fun treeUriToRelativePath(uri: Uri): String? {
707	    val documentId = runCatching { DocumentsContract.getTreeDocumentId(uri) }.getOrNull() ?: return null
708	    val parts = documentId.split(":", limit = 2)
709	    if (parts.size < 2) return null
710	    if (parts[0] != "primary") return null
711	    return parts[1].trim('/')
712	}
